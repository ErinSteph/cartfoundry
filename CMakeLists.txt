cmake_minimum_required(VERSION 3.21)
project(CartFoundry VERSION 0.8.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent Test Multimedia)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/mgba/CMakeLists.txt")
    message(FATAL_ERROR "Run cmake -P scripts/fetch-mgba.cmake to fetch the pinned emulator source.")
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(LIBMGBA_ONLY ON)
set(M_CORE_GB ON)
set(M_CORE_GBA ON)
add_subdirectory(third_party/mgba EXCLUDE_FROM_ALL)
set_target_properties(mgba PROPERTIES AUTOMOC OFF)
add_library(cf_emulation editor/src/emulator_core.cpp)
target_link_libraries(cf_emulation PUBLIC Qt6::Core Qt6::Gui PRIVATE mgba)
target_include_directories(cf_emulation PUBLIC editor/src)
target_compile_definitions(cf_emulation PRIVATE BUILD_STATIC)
add_library(cf_project editor/src/project.cpp editor/src/map_authoring.cpp editor/src/screen_model.cpp editor/src/builder.cpp editor/src/assets.cpp editor/src/behavior.cpp editor/src/world_model.cpp editor/src/pokemon_example.cpp editor/src/game_model.cpp editor/src/game_script.cpp editor/src/game_session.cpp runtime/cf_flow.c runtime/cf_world.c runtime/cf_behavior.c runtime/cf_animation.c runtime/cf_game.c runtime/cf_sound.c runtime/cf_raycast.c)
target_include_directories(cf_project PUBLIC editor/src runtime)
target_link_libraries(cf_project PUBLIC Qt6::Core Qt6::Gui Qt6::Multimedia)
target_compile_definitions(cf_project PRIVATE CF_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
qt_add_executable(CartFoundry MACOSX_BUNDLE WIN32 editor/src/main.cpp editor/src/window.cpp editor/src/screen_builder.cpp editor/src/script_workspace.cpp editor/src/graph.cpp editor/src/world_editor.cpp editor/src/map_objects.cpp editor/src/importer.cpp editor/src/emulator_widget.cpp editor/src/script_editor.cpp editor/src/animation_editor.cpp editor/src/data_editor.cpp editor/src/pixel_editor.cpp editor/src/event_tools.cpp)
target_link_libraries(CartFoundry PRIVATE cf_project cf_emulation Qt6::Widgets Qt6::Concurrent)
set_target_properties(CartFoundry PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER cat.erin.cartfoundry MACOSX_BUNDLE_BUNDLE_NAME CartFoundry MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
add_executable(cf-build editor/src/cli.cpp)
target_link_libraries(cf-build PRIVATE cf_project)
enable_testing()
add_executable(cf-tests tests/project_tests.cpp)
target_link_libraries(cf-tests PRIVATE cf_project Qt6::Test)
target_compile_definitions(cf-tests PRIVATE CF_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
add_test(NAME project-and-build COMMAND cf-tests)
add_executable(cf-flow-tests tests/flow_tests.c runtime/cf_flow.c)
target_include_directories(cf-flow-tests PRIVATE runtime)
add_test(NAME runtime-flow COMMAND cf-flow-tests)
add_executable(cf-world-tests tests/world_tests.c runtime/cf_world.c)
target_include_directories(cf-world-tests PRIVATE runtime)
add_test(NAME runtime-world COMMAND cf-world-tests)
add_executable(cf-emulator-tests tests/emulator_tests.cpp)
target_link_libraries(cf-emulator-tests PRIVATE cf_project cf_emulation Qt6::Test)
add_test(NAME emulator-integration COMMAND cf-emulator-tests)
add_executable(cf-behavior-tests tests/behavior_tests.c runtime/cf_behavior.c runtime/cf_world.c)
target_include_directories(cf-behavior-tests PRIVATE runtime)
add_test(NAME runtime-behavior COMMAND cf-behavior-tests)
add_executable(cf-animation-tests tests/animation_tests.c runtime/cf_animation.c)
target_include_directories(cf-animation-tests PRIVATE runtime)
add_test(NAME runtime-animation COMMAND cf-animation-tests)
add_executable(cf-systems-tests tests/systems_tests.cpp)
target_link_libraries(cf-systems-tests PRIVATE cf_project Qt6::Test)
target_compile_definitions(cf-systems-tests PRIVATE CF_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
add_test(NAME game-systems COMMAND cf-systems-tests)
add_executable(cf-pokemon-tests tests/pokemon_tests.cpp)
target_link_libraries(cf-pokemon-tests PRIVATE cf_project cf_emulation Qt6::Test)
add_test(NAME pokemon-adventure COMMAND cf-pokemon-tests)

add_executable(cf-versatility-tests tests/versatility_tests.cpp)
target_link_libraries(cf-versatility-tests PRIVATE cf_project cf_emulation Qt6::Test)
add_test(NAME versatility-and-rendering COMMAND cf-versatility-tests)
add_executable(cf-scene-tests tests/scene_tests.cpp)
target_link_libraries(cf-scene-tests PRIVATE cf_project cf_emulation Qt6::Test)
add_test(NAME scenes-and-startup COMMAND cf-scene-tests)
add_executable(cf-map-authoring-tests tests/map_authoring_tests.cpp)
target_link_libraries(cf-map-authoring-tests PRIVATE cf_project Qt6::Test)
add_test(NAME map-authoring COMMAND cf-map-authoring-tests)
add_executable(cf-map-canvas-tests tests/map_canvas_tests.cpp editor/src/world_editor.cpp editor/src/map_objects.cpp editor/src/importer.cpp editor/src/animation_editor.cpp editor/src/event_tools.cpp)
target_link_libraries(cf-map-canvas-tests PRIVATE cf_project Qt6::Widgets Qt6::Test)
add_test(NAME map-canvas-interaction COMMAND cf-map-canvas-tests)
set_tests_properties(map-canvas-interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
